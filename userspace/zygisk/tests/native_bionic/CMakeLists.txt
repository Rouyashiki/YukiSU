cmake_minimum_required(VERSION 3.24)
project(YukiSUCoreIntegration LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
add_subdirectory(../../core core)
get_filename_component(CORE ../../core/src ABSOLUTE)
get_filename_component(ROOT ../../../.. ABSOLUTE)
get_filename_component(SDK ../../third_party/yukilinker ABSOLUTE)

add_library(integration_dependency SHARED dependency.c)
target_link_options(integration_dependency PRIVATE -Wl,-z,max-page-size=16384)
add_library(integration_host SHARED dependency.c)
target_compile_definitions(integration_host PRIVATE HOST_DEPENDENCY=1)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/host.map"
    "INTEGRATION_HOST { global: integration_system_probe; integration_scope_probe; integration_invoke; local: *; };\n")
target_link_options(integration_host PRIVATE
    "-Wl,--version-script=${CMAKE_CURRENT_BINARY_DIR}/host.map"
    -Wl,-z,max-page-size=16384)
set_property(TARGET integration_host APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_CURRENT_BINARY_DIR}/host.map")
add_library(integration_foreign SHARED dependency.c)
target_compile_definitions(integration_foreign PRIVATE HOST_DEPENDENCY=1 HOST_VALUE=999)
target_link_libraries(integration_foreign PRIVATE dl)
target_link_options(integration_foreign PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_BINARY_DIR}/host.map"
    -Wl,-z,max-page-size=16384)
add_library(integration_duplicate_libc SHARED dependency.c)
target_compile_definitions(integration_duplicate_libc PRIVATE DUPLICATE_LIBC=1)
target_link_options(integration_duplicate_libc PRIVATE -nostdlib -Wl,--as-needed -Wl,-soname,libc.so
    -Wl,-z,max-page-size=16384)
set_target_properties(integration_duplicate_libc PROPERTIES NO_SONAME TRUE)
foreach(kind private private_ifunc emutls missing invalid null_entry)
    add_library(integration_${kind} SHARED module.cpp)
    target_include_directories(integration_${kind} PRIVATE ../../include)
    target_compile_options(integration_${kind} PRIVATE -UNDEBUG)
    target_compile_options(integration_${kind} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
    target_link_libraries(integration_${kind} PRIVATE dl)
    target_link_options(integration_${kind} PRIVATE -Wl,-z,max-page-size=16384)
    if(kind STREQUAL emutls)
        target_compile_options(integration_${kind} PRIVATE -femulated-tls)
    else()
        target_compile_options(integration_${kind} PRIVATE -fno-emulated-tls)
    endif()
    if(kind STREQUAL missing)
        target_compile_definitions(integration_${kind} PRIVATE MISSING_ENTRY=1)
    endif()
    if(kind STREQUAL invalid)
        target_compile_definitions(integration_${kind} PRIVATE INVALID_VERSION=1)
    endif()
    if(kind STREQUAL null_entry)
        target_compile_definitions(integration_${kind} PRIVATE NULL_ENTRY=1)
    endif()
    if(kind STREQUAL private OR kind STREQUAL private_ifunc)
        target_compile_definitions(integration_${kind} PRIVATE PRIVATE_DEPENDENCY=1)
        target_link_libraries(integration_${kind} PRIVATE integration_dependency)
    endif()
    if(kind STREQUAL private_ifunc)
        target_compile_definitions(integration_${kind} PRIVATE NATIVE_IFUNC=1)
    endif()
endforeach()

get_target_property(ordinary_sources zygisk SOURCES)
set(driver_sources ordinary_core.cpp ordinary_hooks.cpp exit_cleanup_bionic.cpp)
foreach(source IN LISTS ordinary_sources)
    if(source STREQUAL "src/core.cpp" OR source STREQUAL "src/hook.cpp" OR source MATCHES "/yukilinker/")
        continue()
    endif()
    get_filename_component(absolute "${source}" ABSOLUTE BASE_DIR "${CORE}/..")
    list(APPEND driver_sources "${absolute}")
endforeach()
add_library(integration_ordinary_core SHARED ${driver_sources})
target_compile_definitions(integration_ordinary_core PRIVATE YUKIZYGISK_LOG_SOURCE=1)
target_include_directories(integration_ordinary_core PRIVATE ${ROOT} ${CORE} ../../include)
target_compile_options(integration_ordinary_core PRIVATE -UNDEBUG -fexceptions -fno-rtti
    -fvisibility=hidden -fno-stack-protector -fno-c++-static-destructors)
target_link_libraries(integration_ordinary_core PRIVATE YukiPLT::yukiplt dl)
yukilinker_add_runtime_sources(integration_ordinary_core)
target_link_options(integration_ordinary_core PRIVATE -Wl,-Bsymbolic-functions
    -Wl,--wrap=yz_patch_text_checked
    -Wl,--wrap=yz_log
    -Wl,--exclude-libs,ALL -Wl,-z,now -Wl,-z,defs -Wl,-z,max-page-size=16384
    -Wl,--wrap=pthread_key_create -Wl,--wrap=pthread_key_delete -Wl,--wrap=fork
    -Wl,--wrap=dl_iterate_phdr -Wl,--wrap=dl_unwind_find_exidx)

add_executable(handoff_bionic handoff_bionic.cpp registers.S)
target_include_directories(handoff_bionic PRIVATE ${ROOT} ${SDK}/include)
target_compile_options(handoff_bionic PRIVATE -UNDEBUG)
target_link_libraries(handoff_bionic PRIVATE dl)

add_executable(native_phdr native_phdr.cpp)
target_include_directories(native_phdr PRIVATE ${ROOT})
target_compile_options(native_phdr PRIVATE -UNDEBUG)
target_link_libraries(native_phdr PRIVATE dl)
add_dependencies(native_phdr integration_dependency)

add_executable(native_empty native_empty.c)
target_include_directories(native_empty PRIVATE ${ROOT})
target_compile_options(native_empty PRIVATE -ffreestanding -fno-builtin -fno-stack-protector)
if(CMAKE_ANDROID_ARCH_ABI STREQUAL "armeabi-v7a")
    target_compile_options(native_empty PRIVATE -marm)
endif()
target_link_options(native_empty PRIVATE -nostdlib -Wl,--as-needed -Wl,-e,_start
    -Wl,--export-dynamic -Wl,-z,max-page-size=16384)
